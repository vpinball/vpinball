# test runner CMake file to be included in main project CMakeLists
#
# vpx-test is a doctest based test runner that shares the application sources but provides
# its own entry point, like the MSVC make/vpx-test.vcxproj which imports vpx-core.vcxitems
# and adds the tests/ sources.
#
# The target is excluded from the default build; build it explicitly:
#   cmake --build <build-dir> --target vpx-test
#
# tests/assets is deployed in a 'test-assets' folder next to the executable, which is where
# the test runner looks for it (FileLocator::AppSubFolder::Root / "test-assets").

# Win32 Editor build (BGFX / GL / DX9)

if(PLATFORM STREQUAL "windows")

   set(VPX_TEST_APP_SOURCES ${VPX_WINDOWS_SOURCES} ${VPX_TEST_SOURCES})
   list(REMOVE_ITEM VPX_TEST_APP_SOURCES src/core/main.cpp)

   add_executable(vpx-test WIN32 EXCLUDE_FROM_ALL
      ${VPX_TEST_APP_SOURCES}
   )

   target_precompile_headers(vpx-test PUBLIC
      src/core/stdafx.h
   )

   if(RENDERER STREQUAL "BGFX")
      target_include_directories(vpx-test PUBLIC
         ${CMAKE_CURRENT_SOURCE_DIR}
         ${CMAKE_CURRENT_BINARY_DIR}/$(IntDir)
         third-party/include
         third-party/include/win32xx/include
         third-party/include/compat/msvc
         src
         plugins
      )
      target_compile_definitions(vpx-test PRIVATE
         ENABLE_BGFX
         $<IF:$<CONFIG:Debug>,BX_CONFIG_DEBUG=1,BX_CONFIG_DEBUG=0>
      )
   elseif(RENDERER STREQUAL "GL")
      target_sources(vpx-test PRIVATE
         src/renderer/DXGIRegistry.h
      )
      target_include_directories(vpx-test PUBLIC
         $ENV{DXSDK_DIR}/Include
         ${CMAKE_CURRENT_SOURCE_DIR}
         ${CMAKE_CURRENT_BINARY_DIR}/$(IntDir)
         third-party/include
         third-party/include/win32xx/include
         src
         plugins
      )
      target_compile_definitions(vpx-test PRIVATE
         ENABLE_OPENGL
      )
   else()
      target_sources(vpx-test PRIVATE
         src/shaders/hlsl_glsl/BallShader.hlsl
         src/shaders/hlsl_glsl/BasicShader.hlsl
         src/shaders/hlsl_glsl/DMDShader.hlsl
         src/shaders/hlsl_glsl/FBShader.hlsl
         src/shaders/hlsl_glsl/FlasherShader.hlsl
         src/shaders/hlsl_glsl/LightShader.hlsl
         src/shaders/hlsl_glsl/UIShader.hlsl
      )
      target_include_directories(vpx-test PUBLIC
         $ENV{DXSDK_DIR}/Include
         ${CMAKE_CURRENT_SOURCE_DIR}
         ${CMAKE_CURRENT_BINARY_DIR}/$(IntDir)
         ${CMAKE_CURRENT_BINARY_DIR}/$(IntDir)/src
         third-party/include
         third-party/include/winsdk
         third-party/include/win32xx/include
         src
         plugins
         shader
      )
      target_compile_definitions(vpx-test PRIVATE
         ENABLE_DX9
      )
   endif()

   target_compile_options(vpx-test PUBLIC
      $<$<CONFIG:Release>:$<$<COMPILE_LANGUAGE:CXX>:${OPT_COMMON}>>
      $<$<CONFIG:Release>:$<$<COMPILE_LANGUAGE:C>:${OPT_COMMON}>>
   )

   if(RENDERER STREQUAL "BGFX")
      target_link_directories(vpx-test PUBLIC
         ${BUILD_LIBS_DIR}
      )
   else()
      target_link_directories(vpx-test PUBLIC
         $ENV{DXSDK_DIR}/Lib/${ARCH}
         ${BUILD_LIBS_DIR}
      )
   endif()

   target_link_libraries(vpx-test
      comctl32.lib
      dxguid.lib
      nvapi.lib
      winmm.lib
      hid.lib
      setupapi.lib
      dinput8.lib
      $<IF:$<STREQUAL:${ARCH},x64>,freeimage64.lib,freeimage.lib>
      $<IF:$<STREQUAL:${ARCH},x64>,SDL364.lib,SDL3.lib>
   )

   if(RENDERER STREQUAL "BGFX")
      target_link_libraries(vpx-test
         $<IF:$<STREQUAL:${ARCH},x64>,openxr_loader64.lib,openxr_loader.lib>
         $<IF:$<STREQUAL:${ARCH},x64>,bgfx64.lib,bgfx.lib>
         $<IF:$<STREQUAL:${ARCH},x64>,bimg64.lib,bimg.lib>
         $<IF:$<STREQUAL:${ARCH},x64>,bimg_encode64.lib,bimg_encode.lib>
         $<IF:$<STREQUAL:${ARCH},x64>,bx64.lib,bx.lib>
      )
   elseif(RENDERER STREQUAL "GL")
      target_link_libraries(vpx-test
         glad
      )
   else()
      target_link_libraries(vpx-test
         d3d9.lib
         d3dx9.lib
         dxerr.lib
      )
   endif()

   target_link_options(vpx-test PUBLIC
      $<$<CONFIG:Release>:/INCREMENTAL:NO>
      $<$<CONFIG:Release>:/OPT:REF>
      $<$<CONFIG:Release>:/OPT:ICF>
      $<$<CONFIG:Release>:/LTCG>
      $<$<CONFIG:Release>:/DEBUG:FULL>
      $<$<CONFIG:Release>:/DYNAMICBASE:NO>
      $<IF:$<STREQUAL:${ARCH},x64>,$<$<CONFIG:Release>:/SAFESEH:NO>,/SAFESEH:NO>
      $<$<NOT:$<STREQUAL:${ARCH},x64>>:/LARGEADDRESSAWARE>
   )

   if(RENDERER STREQUAL "BGFX")
      set(TEST_RUNTIME_OUTPUT_NAME "$<IF:$<STREQUAL:${ARCH},x64>,VPXTest_BGFX64,VPXTest_BGFX>")
   elseif(RENDERER STREQUAL "GL")
      set(TEST_RUNTIME_OUTPUT_NAME "$<IF:$<STREQUAL:${ARCH},x64>,VPXTest_GL64,VPXTest_GL>")
   else()
      set(TEST_RUNTIME_OUTPUT_NAME "$<IF:$<STREQUAL:${ARCH},x64>,VPXTest_DX964,VPXTest_DX9>")
   endif()

   set_target_properties(vpx-test PROPERTIES
      RUNTIME_OUTPUT_NAME "${TEST_RUNTIME_OUTPUT_NAME}"
      MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>"
   )

   add_custom_command(TARGET vpx-test POST_BUILD
      COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_SOURCE_DIR}/tests/assets" "$<TARGET_FILE_DIR:vpx-test>/test-assets"
   )

   if(POST_BUILD_COPY_EXT_LIBS)
      add_custom_command(TARGET vpx-test POST_BUILD
         COMMAND "${CMAKE_COMMAND}" -E copy "${RUNTIME_LIBS_DIR}/$<IF:$<STREQUAL:${ARCH},x64>,freeimage64.dll,freeimage.dll>" "$<TARGET_FILE_DIR:vpx-test>"
         COMMAND "${CMAKE_COMMAND}" -E copy "${RUNTIME_LIBS_DIR}/$<IF:$<STREQUAL:${ARCH},x64>,SciLexerVP64.dll,SciLexerVP.dll>" "$<TARGET_FILE_DIR:vpx-test>"
         COMMAND "${CMAKE_COMMAND}" -E copy "${RUNTIME_LIBS_DIR}/$<IF:$<STREQUAL:${ARCH},x64>,SDL364.dll,SDL3.dll>" "$<TARGET_FILE_DIR:vpx-test>"
      )
      if(RENDERER STREQUAL "BGFX")
         add_custom_command(TARGET vpx-test POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E copy "${RUNTIME_LIBS_DIR}/$<IF:$<STREQUAL:${ARCH},x64>,openxr_loader64.dll,openxr_loader.dll>" "$<TARGET_FILE_DIR:vpx-test>"
         )
      elseif(RENDERER STREQUAL "DX9")
         add_custom_command(TARGET vpx-test POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E copy "${RUNTIME_LIBS_DIR}/$<IF:$<STREQUAL:${ARCH},x64>,D3DX9_4364.dll,D3DX9_4332.dll>" "$<TARGET_FILE_DIR:vpx-test>"
         )
      endif()
   endif()
