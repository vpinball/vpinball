
# plugin CMake file to be included in main project CMakelists
# environment settings such as CMAKE_SYSTEM_NAME and CMAKE_OSX_SYSROOT should be already configured by root CMakeFile.txt

set(BUILD_SHARED ON)
set(BUILD_STATIC OFF)

if(PLATFORM STREQUAL "ios" OR PLATFORM STREQUAL "ios-simulator" OR PLATFORM STREQUAL "tvos" OR PLATFORM STREQUAL "android")
   set(BUILD_SHARED OFF)
   set(BUILD_STATIC ON)
endif()

if(BUILD_STATIC)
   if(DEFINED PLUGINS_DIR)
      message(WARNING "PLUGINS_DIR is ignored on ${PLATFORM}; plugins are static libraries linked into the application")
   endif()
elseif(BUILD_APP)
   if(DEFINED PLUGINS_DIR)
      message(WARNING "PLUGINS_DIR is ignored when BUILD_APP is ON; plugins are staged next to the application")
   endif()
   if(PLATFORM STREQUAL "macos")
      set(PLUGINS_DIR "$<TARGET_FILE_DIR:vpinball>/../PlugIns")
   else()
      set(PLUGINS_DIR "$<TARGET_FILE_DIR:vpinball>/plugins")
   endif()
else()
   if(NOT DEFINED PLUGINS_DIR)
      set(PLUGINS_DIR "${CMAKE_BINARY_DIR}/plugins")
   endif()
   cmake_path(ABSOLUTE_PATH PLUGINS_DIR BASE_DIRECTORY "${CMAKE_BINARY_DIR}" NORMALIZE)
   message(STATUS "PLUGINS_DIR: ${PLUGINS_DIR}")
   set(PLUGINS_DIR "$<1:${PLUGINS_DIR}>")
endif()

if(PLATFORM STREQUAL "macos")
   set(CMAKE_INSTALL_RPATH "@loader_path/../../Frameworks;@loader_path;@executable_path")
endif()
